-- git diff del archivo SIN atributo:
     sin-atributo.ps1 | Bin 56 -> 68 bytes
     1 file changed, 0 insertions(+), 0 deletions(-)
    Binary files a/sin-atributo.ps1 and b/sin-atributo.ps1 differ
-- git diff del archivo CON working-tree-encoding=UTF-16LE:
    -Write-Host "Configuración"
    +Write-Host "Configuración nueva"
-- índice del archivo con atributo: UTF-8 válido; working tree: NO es UTF-8
