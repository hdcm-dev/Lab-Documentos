   | * text=auto eol=lf
   | *.{cmd,bat} text eol=crlf

     a.cmd: eol: lf
     b.BAT: eol: lf
     c.bat: eol: lf
     d.sh: eol: lf
   contenedor:
     (git version 2.39.5)
     a.cmd: eol: lf

     d.sh: eol: lf

## 2. Un patrón por extensión (con clases para mayúsculas)

   | *.bat text eol=crlf
   | *.[cC][mM][dD] text eol=crlf
   | *.[bB][aA][tT] text eol=crlf

     a.cmd: eol: crlf
     b.BAT: eol: crlf
     c.bat: eol: crlf
     d.sh: eol: lf

     (git version 2.39.5)
